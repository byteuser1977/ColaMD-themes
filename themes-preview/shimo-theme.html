def generate_meeting_minutes(attendees, topics, decisions):
    """
    生成会议纪要
    """
    minutes = {
        "date": datetime.now(),
        "attendees": attendees,
        "topics": topics,
        "decisions": decisions,
        "action_items": []
    }
    return minutes
